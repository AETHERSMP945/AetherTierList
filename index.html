import json import os from datetime import datetime class AetherMC: def __init__(self, title="AetherMC Tier List"): self.title = title self.version = "1.0" self.created = datetime.now().strftime("%Y-%m-%d") self.tiers = { "S": [], "A": [], "B": [], "C": [], "D": [], "F": [] } def add(self, item: str, tier: str = "B"): tier = tier.upper().strip() if tier not in self.tiers: print("❌ Invalid tier! Use S, A, B, C, D, or F.") return False item = item.strip() if not item: print("❌ Item name cannot be empty.") return False if item not in self.tiers[tier]: self.tiers[tier].append(item) print(f"✅ Added '{item}' → {tier} Tier") return True else: print(f"⚠️ '{item}' is already in {tier} Tier") return False def remove(self, item: str): for tier in self.tiers: if item in self.tiers[tier]: self.tiers[tier].remove(item) print(f"🗑 Removed '{item}' from {tier} Tier") return True print(f"❌ '{item}' not found.") return False def move(self, item: str, new_tier: str): self.remove(item) self.add(item, new_tier) def display(self): print("\n" + "═" * 60) print(f" {self.title.upper()}") print(f" v{self.version} - {self.created}") print("═" * 60 + "\n") tier_order = ["S", "A", "B", "C", "D", "F"] colors = {"S": "🔥", "A": "💎", "B": "⭐", "C": "✅", "D": "⚠️", "F": "💀"} for tier in tier_order: items = sorted(self.tiers.get(tier, [])) if items: print(f"{colors[tier]} {tier} Tier ({len(items)})") for item in items: print(f" • {item}") print() def save(self): os.makedirs("data/tierlists", exist_ok=True) data = { "title": self.title, "version": self.version, "created": self.created, "tiers": self.tiers } with open("data/tierlists/aether_save.json", 'w', encoding='utf-8') as f: json.dump(data, f, indent=2) print("💾 Saved to data/tierlists/aether_save.json") def load(self): try: with open("data/tierlists/aether_save.json", 'r', encoding='utf-8') as f: data = json.load(f) self.title = data.get("title", self.title) self.tiers = data.get("tiers", self.tiers) print("✅ Loaded previous save!") return True except FileNotFoundError: print("📁 No save file yet. Starting fresh.") return False # ==================== RUN THE PROGRAM ==================== if __name__ == "__main__": print("✦ Welcome to AetherMC ✦\n") aether = AetherMC("AetherMC - My Minecraft Tier List") aether.load() while True: print("\n" + "─" * 50) print("1. Show Tier List") print("2. Add Item") print("3. Move Item") print("4. Remove Item") print("5. Save") print("6. Exit") print("─" * 50) choice = input("\nChoose option (1-6): ").strip() if choice == "1": aether.display() elif choice == "2": item = input("Item Name: ").strip() tier = input("Tier (S/A/B/C/D/F): ").strip() aether.add(item, tier) elif choice == "3": item = input("Item Name: ").strip() tier = input("New Tier: ").strip() aether.move(item, tier) elif choice == "4": item = input("Item Name to remove: ").strip() aether.remove(item) elif choice == "5": aether.save() elif choice == "6": print("👋 Thanks for using AetherMC!") break else: print("Invalid choice, try again.")
